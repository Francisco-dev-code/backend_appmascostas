package com.tuapp.di

import android.content.Context
import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.Preferences
import androidx.datastore.preferences.preferencesDataStore
import com.tuapp.data.remote.AuthService
import com.tuapp.data.remote.PetService
import dagger.Module
import dagger.Provides
import dagger.hilt.InstallIn
import dagger.hilt.android.qualifiers.ApplicationContext
import dagger.hilt.components.SingletonComponent
import okhttp3.OkHttpClient
import okhttp3.logging.HttpLoggingInterceptor
import retrofit2.Retrofit
import retrofit2.converter.gson.GsonConverterFactory
import java.util.concurrent.TimeUnit
import javax.inject.Singleton

@Module
@InstallIn(SingletonComponent::class)
object AppModule {
    
    // *** IMPORTANTE: Cambiar según tu entorno ***
    // Emulador: "http://10.0.2.2:8080/"
    // Dispositivo físico: "http://TU_IP:8080/"
    private const val BASE_URL = "http://10.0.2.2:8080/"
    
    @Provides
    @Singleton
    fun provideOkHttpClient(): OkHttpClient {
        return OkHttpClient.Builder()
            .addInterceptor(
                HttpLoggingInterceptor().apply {
                    level = HttpLoggingInterceptor.Level.BODY
                }
            )
            .connectTimeout(30, TimeUnit.SECONDS)
            .readTimeout(30, TimeUnit.SECONDS)
            .writeTimeout(30, TimeUnit.SECONDS)
            .build()
    }
    
    @Provides
    @Singleton
    fun provideRetrofit(okHttpClient: OkHttpClient): Retrofit {
        return Retrofit.Builder()
            .baseUrl(BASE_URL)
            .client(okHttpClient)
            .addConverterFactory(GsonConverterFactory.create())
            .build()
    }
    
    @Provides
    @Singleton
    fun provideAuthService(retrofit: Retrofit): AuthService {
        return retrofit.create(AuthService::class.java)
    }
    
    @Provides
    @Singleton
    fun providePetService(retrofit: Retrofit): PetService {
        return retrofit.create(PetService::class.java)
    }
    
    @Provides
    @Singleton
    fun provideDataStore(
        @ApplicationContext context: Context
    ): DataStore<Preferences> {
        return context.dataStore
    }
}

private val Context.dataStore: DataStore<Preferences> by 
    preferencesDataStore(name = "user_prefs")

6. FLUJO DE AUTENTICACIÓN
================================================================================

PASO 1: REGISTRO
-----------------
Usuario → Pantalla de Registro → Llena formulario
       → App llama a authService.register(user)
       → Backend valida (email único)
       → Si OK: 200, muestra "Registro exitoso"
       → Si email existe: 400, muestra "Email ya registrado"


PASO 2: LOGIN
--------------
Usuario → Pantalla de Login → Ingresa email/password
       → App llama a authService.login(LoginRequest)
       → Backend valida credenciales
       → Si OK: Devuelve token
       → App guarda token en DataStore
       → Navega a Home


PASO 3: PERSISTENCIA (DataStoreManager)
----------------------------------------

class DataStoreManager @Inject constructor(
    private val dataStore: DataStore<Preferences>
) {
    companion object {
        private val TOKEN_KEY = stringPreferencesKey("auth_token")
    }
    
    suspend fun saveToken(token: String) {
        dataStore.edit { prefs ->
            prefs[TOKEN_KEY] = token
        }
    }
    
    suspend fun getToken(): String? {
        return dataStore.data.map { prefs ->
            prefs[TOKEN_KEY]
        }.first()
    }
    
    suspend fun clearSession() {
        dataStore.edit { it.clear() }
    }
    
    fun isLoggedIn(): Flow<Boolean> {
        return dataStore.data.map { prefs ->
            prefs[TOKEN_KEY] != null
        }
    }
}


PASO 4: AUTO-LOGIN
-------------------
App inicia → Verifica DataStore
           → Si hay token: Va directo a Home
           → Si no hay token: Muestra Login

7. FLUJO CRUD DE MASCOTAS
================================================================================

A) LISTAR MASCOTAS (Home Screen)
---------------------------------
1. Usuario logueado entra a Home
2. ViewModel llama a petRepository.getPets()
3. Backend ejecuta: SELECT * FROM PETS
4. Devuelve lista JSON
5. App muestra en RecyclerView o LazyColumn


B) AGREGAR MASCOTA
------------------
1. Usuario presiona botón "+" o "Agregar"
2. Muestra formulario (nombre, tipo)
3. Usuario llena y presiona "Guardar"
4. ViewModel llama a petRepository.addPet(Pet(name="Firulais", type="Perro"))
5. Backend inserta en BD: INSERT INTO PETS...
6. Devuelve la mascota con ID generado
7. App actualiza la lista local


C) ACTUALIZAR MASCOTA
----------------------
1. Usuario selecciona una mascota
2. Muestra formulario con datos actuales
3. Usuario modifica y presiona "Actualizar"
4. ViewModel llama a petRepository.updatePet(id=1, pet=PetModificado)
5. Backend actualiza: UPDATE PETS SET name=?, type=? WHERE id=1
6. Devuelve mascota actualizada
7. App actualiza la lista


D) ELIMINAR MASCOTA
-------------------
1. Usuario hace swipe o presiona "Eliminar"
2. Confirma acción
3. ViewModel llama a petRepository.deletePet(id=1)
4. Backend elimina: DELETE FROM PETS WHERE id=1
5. Devuelve 200 OK
6. App remueve de la lista local


EJEMPLO DE VIEWMODEL:
----------------------

@HiltViewModel
class PetViewModel @Inject constructor(
    private val petRepository: PetRepository
) : ViewModel() {
    
    private val _pets = MutableStateFlow<List<Pet>>(emptyList())
    val pets: StateFlow<List<Pet>> = _pets.asStateFlow()
    
    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading.asStateFlow()
    
    fun loadPets() {
        viewModelScope.launch {
            _isLoading.value = true
            petRepository.getPets().onSuccess { list ->
                _pets.value = list
            }.onFailure { error ->
                // Manejar error
            }
            _isLoading.value = false
        }
    }
    
    fun addPet(name: String, type: String) {
        viewModelScope.launch {
            petRepository.addPet(Pet(name = name, type = type))
                .onSuccess {
                    loadPets() // Recargar lista
                }
        }
    }
    
    fun deletePet(id: Long) {
        viewModelScope.launch {
            petRepository.deletePet(id).onSuccess {
                loadPets() // Recargar lista
            }
        }
    }
}

9. TROUBLESHOOTING (SOLUCIÓN DE PROBLEMAS)
================================================================================

PROBLEMA: "Unable to resolve host" o "Failed to connect"
---------------------------------------------------------
CAUSA: Backend no está corriendo o URL incorrecta
SOLUCIÓN:
1. Verifica que el backend esté corriendo (gradlew.bat bootRun)
2. Verifica la URL en AppModule: "http://10.0.2.2:8080/"
3. Si usas dispositivo físico, cambia a tu IP local
4. Verifica que el firewall no bloquee el puerto 8080


PROBLEMA: "Cleartext HTTP traffic not permitted"
-------------------------------------------------
CAUSA: Android bloquea HTTP por defecto (requiere HTTPS)
SOLUCIÓN:
Agregar en AndroidManifest.xml:
<application android:usesCleartextTraffic="true" ...>


PROBLEMA: "NetworkOnMainThreadException"
-----------------------------------------
CAUSA: Llamadas de red en el hilo principal
SOLUCIÓN:
Usar suspend functions con viewModelScope.launch { }
Todas las funciones de Retrofit deben ser suspend


PROBLEMA: Error 400 Bad Request al registrar
---------------------------------------------
CAUSA: Email ya existe en la base de datos
SOLUCIÓN:
1. Es comportamiento esperado (validación del backend)
2. Usar otro email
3. O reiniciar el backend (borra la BD en memoria)


PROBLEMA: Error 401 Unauthorized en login
------------------------------------------
CAUSA: Email o contraseña incorrectos
SOLUCIÓN:
1. Verifica las credenciales
2. Asegúrate de haber registrado el usuario primero


PROBLEMA: Error 404 Not Found
------------------------------
CAUSA: Ruta incorrecta en Retrofit
SOLUCIÓN:
Verifica que las rutas coincidan:
- Backend: @PostMapping("/auth/register")
- Android: @POST("auth/register")


PROBLEMA: La lista de mascotas no se actualiza
-----------------------------------------------
CAUSA: No se recarga la lista después de agregar/eliminar
SOLUCIÓN:
Llamar a loadPets() después de cada operación exitosa:

fun addPet(name: String, type: String) {
    viewModelScope.launch {
        petRepository.addPet(Pet(name = name, type = type))
            .onSuccess {
                loadPets() // ← Importante
            }
    }
}


PROBLEMA: App se cierra al abrir
---------------------------------
CAUSA POSIBLE 1: Falta @HiltAndroidApp en Application
SOLUCIÓN: Agregar @HiltAndroidApp class TuApplication : Application()

CAUSA POSIBLE 2: Falta permiso INTERNET
SOLUCIÓN: Agregar <uses-permission android:name="android.permission.INTERNET" />

CAUSA POSIBLE 3: Error en el modelo de datos
SOLUCIÓN: Verificar que los modelos coincidan con el backend


PROBLEMA: Backend se reinicia solo
-----------------------------------
CAUSA: Spring Boot DevTools activo
SOLUCIÓN:
Es normal. Reinicia automáticamente al detectar cambios.
Para detenerlo: Ctrl+C en la terminal


PROBLEMA: "Connection timeout"
-------------------------------
CAUSA: Backend tarda mucho en responder
SOLUCIÓN:
Aumentar timeout en OkHttpClient:
.connectTimeout(60, TimeUnit.SECONDS)
.readTimeout(60, TimeUnit.SECONDS)
